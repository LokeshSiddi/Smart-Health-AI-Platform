# RabbitMQ (Automatically removes the container when it stops)
docker run -it --rm --name rabbitmq -p 5672:5672 -p 15672:15672 rabbitmq:4-management

# Keycloak (Automatically removes the container when it stops)
docker run --rm -p 8181:8080 --name keycloak -e KC_BOOTSTRAP_ADMIN_USERNAME=admin -e KC_BOOTSTRAP_ADMIN_PASSWORD=admin quay.io/keycloak/keycloak:26.3.5 start-dev
